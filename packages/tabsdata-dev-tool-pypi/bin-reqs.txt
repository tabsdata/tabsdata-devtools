# bin-reqs.txt -- the external binaries `tools-verify.sh` checked before anyone
# trusted a release, now declared so `tdd --reqs` reports them and a missing one
# refuses the command instead of failing halfway through a publish.
#
# The tool itself shells out to `git` alone (git.py: the prod lane's committed +
# pushed gate); everything else goes through boto3. The rest are here because
# the workflow around the tool needs them -- provisioning with tofu, checking a
# certificate and the DNS delegation with openssl and dig, poking the index with
# curl and jq, building an env with uv -- and finding that out at step 6 of a
# 10-step publish is worse than finding it out now. `aws-check` reports each
# one's version.
#
# No alternatives here: one bare name per line, all required. `tofu` and not
# `terraform` because lanes.yaml is generated by `tofu output`.
[required]
aws
tofu
uv
git
curl
jq
openssl
dig
